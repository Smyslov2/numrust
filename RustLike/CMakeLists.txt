cmake_minimum_required(VERSION 3.10)
project(RustLike CXX)
set(CMAKE_CXX_STANDARD 17)
add_executable(RustLike WIN32 main.cpp)
target_link_libraries(RustLike opengl32 gdi32 user32)
if(MSVC)
  target_compile_options(RustLike PRIVATE /O2)
else()
  target_compile_options(RustLike PRIVATE -O2)
  target_link_options(RustLike PRIVATE -static -mwindows)
endif()
